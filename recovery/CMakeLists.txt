include(ExternalProject)
find_program(RZ_MAKE NAMES gmake make REQUIRED)
set(SODIUM_PREFIX "${CMAKE_CURRENT_BINARY_DIR}/sodium-install")
set(SODIUM_CFLAGS "-O2 -fPIC")
set(SODIUM_LDFLAGS "")
if(APPLE AND CMAKE_OSX_DEPLOYMENT_TARGET)
  string(APPEND SODIUM_CFLAGS " -mmacosx-version-min=${CMAKE_OSX_DEPLOYMENT_TARGET}")
endif()
if(APPLE AND CMAKE_OSX_ARCHITECTURES)
  foreach(arch IN LISTS CMAKE_OSX_ARCHITECTURES)
    string(APPEND SODIUM_CFLAGS " -arch ${arch}")
  endforeach()
endif()
if(RZ_SANITIZE)
  string(APPEND SODIUM_CFLAGS " -fsanitize=address,undefined -fno-omit-frame-pointer")
  string(APPEND SODIUM_LDFLAGS " -fsanitize=address,undefined")
endif()
if(RZ_THREAD_SANITIZE)
  string(APPEND SODIUM_CFLAGS " -fsanitize=thread -fno-omit-frame-pointer")
  string(APPEND SODIUM_LDFLAGS " -fsanitize=thread")
endif()
ExternalProject_Add(sodium_build
  URL "${CMAKE_CURRENT_SOURCE_DIR}/vendor/libsodium-1.0.22.tar.gz"
  URL_HASH SHA256=adbdd8f16149e81ac6078a03aca6fc03b592b89ef7b5ed83841c086191be3349
  CONFIGURE_COMMAND ${CMAKE_COMMAND} -E env "CC=${CMAKE_C_COMPILER}" "CFLAGS=${SODIUM_CFLAGS}" "LDFLAGS=${SODIUM_LDFLAGS}"
    <SOURCE_DIR>/configure --prefix=${SODIUM_PREFIX} --disable-shared --enable-static --with-pic --disable-dependency-tracking
  BUILD_COMMAND ${RZ_MAKE} -j4
  INSTALL_COMMAND ${RZ_MAKE} install
  BUILD_BYPRODUCTS "${SODIUM_PREFIX}/lib/libsodium.a"
  LOG_CONFIGURE ON LOG_BUILD ON LOG_INSTALL ON)
file(MAKE_DIRECTORY "${SODIUM_PREFIX}/include")
add_library(sodium STATIC IMPORTED GLOBAL)
set_target_properties(sodium PROPERTIES IMPORTED_LOCATION "${SODIUM_PREFIX}/lib/libsodium.a"
  INTERFACE_INCLUDE_DIRECTORIES "${SODIUM_PREFIX}/include")
add_dependencies(sodium sodium_build)
